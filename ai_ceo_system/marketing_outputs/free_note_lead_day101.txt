■ 3. スプレッドシート自動処理＆メール通知 GAS 完成コード
スプレッドシート上部メニューの「拡張機能」＞「Apps Script」を開き、下記のコードを貼り付けて保存してください。トリガー（毎日午前8時など）を設定すれば、確認漏れ・処理完了が自動でメール通知されます。

```javascript
function runAutoWorkflow() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const apiKey = PropertiesService.getScriptProperties().getProperty('OPENAI_API_KEY');
  
  for (let i = 1; i < data.length; i++) {
    const [id, type, clause, risk, proposal, status] = data[i];
    if (status === '完了' || !clause) continue;

    try {
      const prompt = `以下の条項をレビューしてください。\n\n条項：${clause}`;
      const response = callOpenAI(apiKey, prompt); // OpenAI APIを呼び出す共通関数（別途定義想定）
      
      // 応答を解析してシートに書き込み（簡易的に分割処理）
      const riskLevel = response.match(/リスク判定：(.*)/)?.[1] || '不明';
      const suggestion = response.match(/修正案および法的根拠：([\s\S]*)/)?.[1] || '解析失敗';

      sheet.getRange(i + 1, 4).setValue(riskLevel);
      sheet.getRange(i + 1, 5).setValue(suggestion);
      sheet.getRange(i + 1, 6).setValue('完了');

      MailApp.sendEmail({
        to: Session.getActiveUser().getEmail(),
        subject: `【契約書チェック完了】ID:${id}`,
        body: `条項のチェックが完了しました。リスク判定：${riskLevel}\n\n修正案：\n${suggestion}`
      });
    } catch (e) {
      sheet.getRange(i + 1, 6).setValue('エラー: ' + e.message);
      console.error(e);
    }
  }
}

function callOpenAI(apiKey, prompt) {
  const url = 'https://api.openai.com/v1/chat/completions';
  const options = {
    method: 'post',
    headers: { 'Authorization': 'Bearer ' + apiKey, 'Content-Type': 'application/json' },
    payload: JSON.stringify({
      model: 'gpt-4-turbo-preview',
      messages: [{ role: 'system', content: 'あなたはプロフェッショナルな法務コンサルタントです。' }, { role: 'user', content: prompt }],
      temperature: 0
    })
  };
  const res = UrlFetchApp.fetch(url, options);
  return JSON.parse(res.getContentText()).choices[0].message.content;
}
```
