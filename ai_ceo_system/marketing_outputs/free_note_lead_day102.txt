■ 3. スプレッドシート自動処理＆メール通知 GAS 完成コード
スプレッドシート上部メニューの「拡張機能」＞「Apps Script」を開き、下記のコードを貼り付けて保存してください。トリガー（毎日午前8時など）を設定すれば、確認漏れ・処理完了が自動でメール通知されます。

```javascript
function runAutoWorkflow() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const apiKey = PropertiesService.getScriptProperties().getProperty('OPENAI_API_KEY');
  const apiUrl = 'https://api.openai.com/v1/chat/completions';

  for (let i = 1; i < data.length; i++) {
    const [target, benefit, hookType, status] = [data[i][0], data[i][1], data[i][2], data[i][3]];
    if (status !== '未処理') continue;

    try {
      sheet.getRange(i + 1, 4).setValue('実行中');
      const prompt = `ターゲット：${target}\nベネフィット：${benefit}\n方向性：${hookType}`;
      const payload = {
        model: "gpt-4o",
        messages: [{role: "system", content: "プロンプト全文をここにペースト（または別セルから参照）"}, {role: "user", content: prompt}],
        response_format: { type: "json_object" }
      };

      const response = UrlFetchApp.fetch(apiUrl, {
        method: 'post',
        contentType: 'application/json',
        headers: { Authorization: `Bearer ${apiKey}` },
        payload: JSON.stringify(payload)
      });

      const resJson = JSON.parse(response.getContentText());
      const content = JSON.parse(resJson.choices[0].message.content);

      sheet.getRange(i + 1, 5).setValue(content.script);
      sheet.getRange(i + 1, 6).setValue(content.advice);
      sheet.getRange(i + 1, 4).setValue('完了');

    } catch (e) {
      sheet.getRange(i + 1, 4).setValue('エラー');
      console.error(e.toString());
    }
  }
  MailApp.sendEmail(Session.getActiveUser().getEmail(), "台本量産ジョブ完了", "スプレッドシートの台本生成が完了しました。確認してください。");
}
```
