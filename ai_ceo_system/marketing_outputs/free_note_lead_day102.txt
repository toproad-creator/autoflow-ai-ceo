■ 3. スプレッドシート自動処理＆メール通知 GAS 完成コード
スプレッドシート上部メニューの「拡張機能」＞「Apps Script」を開き、下記のコードを貼り付けて保存してください。トリガー（毎日午前8時など）を設定すれば、確認漏れ・処理完了が自動でメール通知されます。

```javascript
function runAutoWorkflow() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getActiveSheet();
    const data = sheet.getDataRange().getValues();
    const header = data.shift();
    
    let totalSales = 0;
    let totalCustomers = 0;
    let processedCount = 0;
    const today = new Date();
    
    for (let i = 0; i < data.length; i++) {
      let row = data[i];
      let rowIndex = i + 2;
      let sales = Number(row[2]);
      let customers = Number(row[3]);
      let status = row[4];

      if (status === '未処理') {
        // バリデーションと集計
        if (sales > 0) {
          totalSales += sales;
          totalCustomers += customers;
          sheet.getRange(rowIndex, 5).setValue('完了');
          sheet.getRange(rowIndex, 6).setValue(Utilities.formatDate(today, 'JST', 'yyyy-MM-dd') + ' 集計済');
          processedCount++;
        } else {
          sheet.getRange(rowIndex, 5).setValue('エラー');
          sheet.getRange(rowIndex, 6).setValue('売上不正値：要確認');
        }
      }
    }

    if (processedCount > 0) {
      const avgSpend = totalCustomers > 0 ? Math.round(totalSales / totalCustomers) : 0;
      const message = `【自動売上速報】\n本日集計数: ${processedCount}件\n総売上: ${totalSales.toLocaleString()}円\n客単価: ${avgSpend.toLocaleString()}円`;
      
      // 通知処理（MailAppを代替利用、実際はUrlFetchAppでSlack等へ）
      MailApp.sendEmail({
        to: Session.getActiveUser().getEmail(),
        subject: '【自動通知】週次売上KPIレポート',
        body: message
      });
      console.log('Workflow completed successfully');
    }
  } catch (e) {
    console.error('Error: ' + e.message);
    MailApp.sendEmail(Session.getActiveUser().getEmail(), '【エラー】売上集計スクリプト', e.message);
  }
}
```
